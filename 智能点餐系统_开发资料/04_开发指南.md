```python
# tools/db_tool.py
from config import Config
from sqlalchemy import create_engine, text


_engine = None


def get_engine():
    global _engine
    if _engine is None:
        _engine = create_engine(Config.get_mysql_url(), pool_pre_ping=True)
    return _engine


def query_user_profile(user_id: str) -> dict:
    """
    根据 user_id 查询用户画像
    返回格式：
    {
        "user_id": "user_001",
        "favorite_cuisines": ["川菜", "湘菜"],
        "taste_preferences": ["辣", "酸"],
        "spicy_tolerance": 3,
        "avg_order_price": 50.0,
        "recent_orders": [...]
    }
    """
    engine = get_engine()
    with engine.connect() as conn:
        # 查用户基本信息
        user_row = conn.execute(
            text("SELECT * FROM users WHERE user_id = :uid"),
            {"uid": user_id}
        ).fetchone()

        if not user_row:
            return None

        # 查历史订单
        orders = conn.execute(
            text("""
                SELECT dish_name, restaurant_name, order_date
                FROM orders
                WHERE user_id = :uid
                ORDER BY order_date DESC
                LIMIT 5
            """),
            {"uid": user_id}
        ).fetchall()

        return {
            "user_id": user_row.user_id,
            "favorite_cuisines": user_row.favorite_cuisines.split(",") if user_row.favorite_cuisines else [],
            "taste_preferences": user_row.taste_preferences.split(",") if user_row.taste_preferences else [],
            "spicy_tolerance": user_row.spicy_tolerance or 3,
            "avg_order_price": float(user_row.avg_order_price or 0),
            "recent_orders": [
                {"dish_name": r.dish_name, "restaurant": r.restaurant_name, "date": str(r.order_date)}
                for r in orders
            ]
        }


def query_dishes_by_taste(taste_tags: list) -> list[dict]:
    """
    根据口味标签查询菜品
    taste_tags: ["辣", "酸"]
    """
    engine = get_engine()
    with engine.connect() as conn:
        # 构建模糊查询条件
        conditions = " OR ".join([f"taste_tags LIKE :tag{i}" for i in range(len(taste_tags))])
        params = {f"tag{i}": f"%{tag}%" for i, tag in enumerate(taste_tags)}

        rows = conn.execute(
            text(f"""
                SELECT dish_id, dish_name, restaurant_id, price, taste_tags, cuisine
                FROM dishes
                WHERE {conditions}
                LIMIT 10
            """),
            params
        ).fetchall()

        return [
            {
                "dish_id": r.dish_id,
                "dish_name": r.dish_name,
                "restaurant_id": r.restaurant_id,
                "price": float(r.price),
                "taste_tags": r.taste_tags,
                "cuisine": r.cuisine,
            }
            for r in rows
        ]


def query_restaurant_by_id(restaurant_id: str) -> dict:
    """根据餐厅ID查询餐厅详情"""
```

```python
# tools/ammap_tool.py
import requests
from config import Config


def get_nearby_restaurants(longitude: float, latitude: float, radius: int = 3000) -> list[dict]:
    """
    调用高德地图 POI 搜索接口，查询附近餐厅
    返回 list[dict]，每个 dict 包含餐厅基础信息
    """
    url = "https://restapi.amap.com/v3/place/around"
    params = {
        "key": Config.AMAP_API_KEY,
        "location": f"{longitude},{latitude}",
        "radius": radius,
        "types": "050000",  # 餐饮服务
        "offset": 10,
        "page": 1,
        "extensions": "all",
    }

    resp = requests.get(url, params=params, timeout=10)
    data = resp.json()

    if data.get("status") != "1":
        return []

    pois = data.get("pois", [])
    result = []
    for poi in pois:
        # 高德返回的坐标是"经度,纬度"，需要拆分
        location = poi.get("location", "").split(",")
        result.append({
            "restaurant_id": poi.get("id", ""),
            "name": poi.get("name", ""),
            "address": poi.get("address", ""),
            "longitude": float(location[0]) if len(location) >= 1 else 0.0,
            "latitude": float(location[1]) if len(location) >= 2 else 0.0,
            "rating": 0.0,  # 高德POI无评分，需结合其他数据
            "avg_price": 0.0,
            "tags": [tag.get("name", "") for tag in poi.get("business_info", {}).get("eating_type", [])],
        })

    return result


def get_distance(
    origin_lng: float, origin_lat: float,
    dest_lng: float, dest_lat: float
) -> dict:
    """
    计算两点间直线距离（调用高德距离测量接口）
    返回 {"distance": 1234, "duration": 600}
    distance: 米, duration: 秒
    """
    url = "https://restapi.amap.com/v3/distance"
    params = {
        "key": Config.AMAP_API_KEY,
        "origins": f"{origin_lng},{origin_lat}",
        "destination": f"{dest_lng},{dest_lat}",
        "type": "1",  # 直線距離
    }

    resp = requests.get(url, params=params, timeout=10)
    data = resp.json()

    if data.get("status") != "1" or not data.get("results"):
        return {"distance": 0, "duration": 0}

    first = data["results"][0]
    return {
        "distance": int(first.get("distance", 0)),
        "duration": int(first.get("duration", 0)),
    }
```

```python
# tools/pinecone_tool.py
import os
from langchain_community.embeddings import DashScopeEmbeddings
from pinecone import Pinecone
from config import Config


# 初始化全局客户端
_pc = None
_embed_model = None


def get_pinecone_client() -> Pinecone:
    global _pc
    if _pc is None:
        _pc = Pinecone(api_key=Config.PINECONE_API_KEY)
    return _pc


def get_embed_model():
    """获取嵌入模型（使用阿里通义文本嵌入）"""
    global _embed_model
    if _embed_model is None:
        _embed_model = DashScopeEmbeddings(
            model="text-embedding-v3",
            dashscope_api_key=Config.DASHSCOPE_API_KEY,
        )
    return _embed_model


def ensure_index_exists():
    """确保 Pinecone 索引存在，不存在则创建"""
    pc = get_pinecone_client()
    index_name = Config.PINECONE_INDEX

    existing = [idx.name for idx in pc.list_indexes()]

    if index_name not in existing:
        pc.create_index(
            name=index_name,
            dimension=1536,  # text-embedding-v3 的维度
            metric="cosine",
        )
    return index_name


def search_similar_dishes(food_name: str, top_k: int = 5) -> list[dict]:
    """
    给定食物名称，检索 Pinecone 中最相似的菜品
    """
    index_name = ensure_index_exists()
    embed_model = get_embed_model()
    index = get_pinecone_client().Index(index_name)

    # 1. 将食物名称转为向量
    vector = embed_model.embed_query(food_name)

    # 2. 向量检索
    results = index.query(
        vector=vector,
        top_k=top_k,
        include_metadata=True,
    )

    # 3. 解析结果
    matches = results.get("matches", [])
    return [
        {
            "dish_id": match["id"],
            "dish_name": match["metadata"].get("dish_name", ""),
            "restaurant_id": match["metadata"].get("restaurant_id", ""),
            "score": match["score"],
        }
        for match in matches
    ]


def upsert_dish_vector(dish_id: str, food_name: str, restaurant_id: str):
    """
    将新菜品入库：生成向量 + 存入 Pinecone
    """
    index_name = ensure_index_exists()
    embed_model = get_embed_model()
    index = get_pinecone_client().Index(index_name)

    vector = embed_model.embed_query(food_name)

    index.upsert(vectors=[
        {
            "id": dish_id,
            "values": vector,
            "metadata": {
                "dish_name": food_name,
                "restaurant_id": restaurant_id,
            }
        }
    ])
```

```python
# tools/llm_tool.py
import dashscope
from dashscope import Generation
from config import Config


dashscope.api_key = Config.DASHSCOPE_API_KEY


def recognize_food_text(text: str) -> dict:
    """
    输入文字描述，返回食物名称和置信度
    """
    prompt = f"""
你是一个专业的食物识别助手。用户输入了一段文字描述，请你识别它描述的是什么食物。

用户输入：{text}

请以JSON格式返回识别结果：
{{
    "food_name": "食物名称",
    "confidence": 0.95,
    "candidates": [
        {{"name": "最可能的食物", "confidence": 0.95}},
        {{"name": "次可能", "confidence": 0.03}}
    ]
}}

注意：
- 只返回JSON，不要有其他内容
- confidence 是 0-1 的小数
- 如果无法确定，返回 confidence=0
"""
    response = Generation.call(
        model="qwen-plus",
        messages=[{"role": "user", "content": prompt}],
        result_format="message",
    )

    if response.status_code != 200:
        return {"food_name": "", "confidence": 0, "candidates": []}

    content = response.output.choices[0].message.content.strip()

    # 简单解析 JSON（生产环境建议用 json.loads 配合正则提取）
    import json, re
    try:
        # 去掉可能的 markdown 代码块
        match = re.search(r"\{.*\}", content, re.DOTALL)
        if match:
            data = json.loads(match.group())
            return data
    except Exception:
        pass

    return {"food_name": "", "confidence": 0, "candidates": []}

    )

    if response.status_code != 200:
        return "推荐生成失败，请稍后重试。"

    return response.output.choices[0].message.content.strip()
```

### 【任务7】服务层 + Agent层
**文件**：`service/diancan_service.py`、`agent/smart_agent.py`

**service/diancan_service.py 模板**：
```python
# service/diancan_service.py
from tools.db_tool import query_user_profile, query_restaurant_by_id
from tools.pinecone_tool import search_similar_dishes
from tools.amap_tool import get_nearby_restaurants, get_distance
from tools.llm_tool import recognize_food_text, analyze_taste, generate_recommendation


def recognize(text: str) -> dict:
    """食物识别"""
    return recognize_food_text(text)


def recommend(
    user_id: str,
    food_name: str,
    longitude: float,
    latitude: float,
    distance: int = 3000,
) -> dict:
    """
    智能推荐主流程：
    1. 分析口味
    2. 查用户画像
    3. 查附近餐厅
    4. 检索相似菜品
    5. 生成推荐理由
    """
    # 1. 口味分析
    taste_info = analyze_taste(food_name)

    # 2. 用户画像（并行）
    user_profile = query_user_profile(user_id) or {}

    # 3. 附近餐厅
    nearby = get_nearby_restaurants(longitude, latitude, distance)

    # 4. 向量检索相似菜品
    similar_dishes = search_similar_dishes(food_name, top_k=5)

    # 5. 组装上下文，生成推荐
    context = {
        "user_id": user_id,
        "food_name": food_name,
        "taste_info": taste_info,
        "user_profile": user_profile,
        "restaurants": nearby,
        "dishes": similar_dishes,
    }

    recommendation_text = generate_recommendation(context)

    return {
        "food_name": food_name,
        "taste_info": taste_info,
        "recommendation": recommendation_text,
        "restaurants": nearby,
    }
```

**agent/smart_agent.py 模板**：
```python
# agent/smart_agent.py
from service.diancan_service import recognize, recommend


def handle_recommend(user_id: str, text: str, longitude: float, latitude: float, distance: int = 3000) -> dict:
    """
    Agent 的入口：处理推荐请求
    决定走什么流程，调用什么工具
    """
    # 1. 识别食物
    recognize_result = recognize(text)
    food_name = recognize_result.get("food_name", "")

    if not food_name:
        return {
            "code": 5001,
            "message": "食物识别失败",
            "data": None
        }

    # 2. 执行推荐
    result = recommend(
        user_id=user_id,
        food_name=food_name,
        longitude=longitude,
        latitude=latitude,
        distance=distance,
    )

    return {
        "code": 0,
        "message": "success",
        "data": result
    }
```
